cmake_minimum_required(VERSION 3.16)
project(book_interfaces)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Readout.msg")
add_executable(roundtrip src/roundtrip.cpp)
target_compile_features(roundtrip PRIVATE cxx_std_14)
ament_target_dependencies(roundtrip rclcpp)
rosidl_get_typesupport_target(book_typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(roundtrip "${book_typesupport}")
install(TARGETS roundtrip DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
